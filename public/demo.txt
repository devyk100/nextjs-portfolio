File {
Grid = "@tdr@"
Plot = "@tdrdat@"
Parameter= "@parameter@"
Current = "@plot@"
Output = "@log@"
ACExtract = "@acplot@" }

Electrode {
{ Name="so" Voltage=0.0  }
{ Name="dr" Voltage=0.0  }
{ Name="ga" Voltage=0.0  }
 }

Thermode {
{ Name = "ga" Temperature = 100}
{ Name = "dr" Temperature = 100}
{ Name = "so" Temperature = 100}
}


Physics(Material="TiN") {
    MetalWorkFunction(WorkFunction=4.45)
}

Physics {
    Fermi
    IncompleteIonization

    eQuantumPotential

    Mobility(
        PhuMob
        DopingDep
        HighFieldSaturation(GradQuasiFermi)
        Enormal
    )

    EffectiveIntrinsicDensity(
        BandGapNarrowing(OldSlotboom)
    )

    Recombination(
        SRH(DopingDependence TempDependence)
        Auger
    )
}
   
Math {
   Extrapolate     Iterations=50     Notdamped =100
   RelErrControl     ErRef(Electron)=1.e10    ErRef(Hole)=1.e10 }


Plot{
   eDensity hDensity   TotalCurrent/Vector eCurrent/Vector hCurrent/Vector
   eMobility hMobility   eVelocity hVelocity ElectricField Potential SpaceCharge
   eQuasiFermi hQuasiFermi

   eTemperature Temperature hTemperature

   ElectricField/Vector Potential SpaceCharge
   FEPolarization

   Doping DonorConcentration AcceptorConcentration

   BandGap    BandGapNarrowing   Affinity
   ConductionBand ValenceBand
   eQuantumPotential hQuantumPotential
}

	Poisson
	Coupled { Poisson  }
	Coupled { Poisson Electron Hole }
	Save(FilePrefix="ZeroBias")

	Coupled { Poisson Electron Hole } }
